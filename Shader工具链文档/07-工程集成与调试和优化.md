## 1. 构建系统集成

### 1.1 目标

| 目标 | 说明 |
|------|------|
| **增量** | 只重编改动的 shader |
| **可重现** | 不同机器/路径产出字节一致 |
| **可并行** | 数百个变体能吃满 CPU |
| **失败可读** | 报错能定位到 shader 源码行 |
| **产物可查** | 生成物能被人检视与 diff |

### 1.2 方案 A：CMake `add_custom_command`（简单直接）

```cmake
# ---------------------------------------------------------------
# ShaderToolchain.cmake
# ---------------------------------------------------------------
find_program(DXC_EXECUTABLE       NAMES dxc       REQUIRED)
find_program(SPIRV_VAL_EXECUTABLE NAMES spirv-val REQUIRED)
find_program(SPIRV_OPT_EXECUTABLE NAMES spirv-opt REQUIRED)
find_program(SPIRV_DIS_EXECUTABLE NAMES spirv-dis REQUIRED)

set(SHADER_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
set(SHADER_OUT_DIR ${CMAKE_CURRENT_BINARY_DIR}/shaders)
file(MAKE_DIRECTORY ${SHADER_OUT_DIR})

# stage 后缀 → DXC profile 与默认入口点
function(_resolve_profile ext out_profile out_entry)
    if(ext STREQUAL ".vert")   set(${out_profile} "vs_6_6" PARENT_SCOPE)
                               set(${out_entry}   "VSMain" PARENT_SCOPE)
    elseif(ext STREQUAL ".frag") set(${out_profile} "ps_6_6" PARENT_SCOPE)
                                 set(${out_entry}   "PSMain" PARENT_SCOPE)
    elseif(ext STREQUAL ".comp") set(${out_profile} "cs_6_6" PARENT_SCOPE)
                                 set(${out_entry}   "CSMain" PARENT_SCOPE)
    elseif(ext STREQUAL ".mesh") set(${out_profile} "ms_6_6" PARENT_SCOPE)
                                 set(${out_entry}   "MSMain" PARENT_SCOPE)
    elseif(ext STREQUAL ".task") set(${out_profile} "as_6_6" PARENT_SCOPE)
                                 set(${out_entry}   "ASMain" PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unknown shader extension: ${ext}")
    endif()
endfunction()

# ---------------------------------------------------------------
# compile_shader(<target> <source> [DEFINES d1 d2 ...] [ENTRY name])
# ---------------------------------------------------------------
function(compile_shader target src)
    cmake_parse_arguments(ARG "" "ENTRY" "DEFINES" ${ARGN})

    get_filename_component(name  ${src} NAME_WE)
    get_filename_component(ext   ${src} EXT)
    _resolve_profile(${ext} profile default_entry)

    if(ARG_ENTRY)
        set(entry ${ARG_ENTRY})
    else()
        set(entry ${default_entry})
    endif()

    # 定义宏集合参与文件名 → 不同变体不互相覆盖
    string(REPLACE ";" "_" def_tag "${ARG_DEFINES}")
    if(def_tag STREQUAL "")
        set(out_stem "${name}${ext}")
    else()
        string(MAKE_C_IDENTIFIER "${def_tag}" def_tag)
        set(out_stem "${name}.${def_tag}${ext}")
    endif()

    set(spv      ${SHADER_OUT_DIR}/${out_stem}.spv)
    set(dis_file ${SHADER_OUT_DIR}/${out_stem}.spv.txt)

    set(cmd_defines "")
    foreach(d ${ARG_DEFINES})
        list(APPEND cmd_defines -D ${d})
    endforeach()

    add_custom_command(
        OUTPUT  ${spv} ${dis_file}
        COMMAND ${CMAKE_COMMAND} -E echo ">> SHADER ${name} [${ARG_DEFINES}]"
        COMMAND ${DXC_EXECUTABLE}
                -spirv
                -fspv-target-env=vulkan1.3
                -T ${profile}
                -E ${entry}
                ${cmd_defines}
                -I ${SHADER_SRC_DIR}
                -Fo ${spv}
                ${src}
        COMMAND ${SPIRV_VAL_EXECUTABLE} --target-env vulkan1.3 ${spv}
        COMMAND ${SPIRV_OPT_EXECUTABLE} -O --strip-nonsemantic -o ${spv} ${spv}
        COMMAND ${SPIRV_DIS_EXECUTABLE} ${spv} > ${dis_file}
        DEPENDS ${src}
        COMMENT "Compiling shader ${out_stem}"
        VERBATIM
    )
    set_property(TARGET ${target} APPEND PROPERTY SHADER_SPV_FILES ${spv})
endfunction()

# ---------------------------------------------------------------
# 批量注册
# ---------------------------------------------------------------
function(target_shaders target)
    file(GLOB_RECURSE sources CONFIGURE_DEPENDS
         ${SHADER_SRC_DIR}/*.hlsl
         ${SHADER_SRC_DIR}/*.vert
         ${SHADER_SRC_DIR}/*.frag
         ${SHADER_SRC_DIR}/*.comp)

    foreach(src ${sources})
        compile_shader(${target} ${src} ${ARGN})
    endforeach()

    get_target_property(spvs ${target} SHADER_SPV_FILES)
    add_custom_target(${target}_shaders ALL DEPENDS ${spvs})
    add_dependencies(${target} ${target}_shaders)
endfunction()
```

使用：

```cmake
add_executable(GameEditor main.cpp)
target_shaders(GameEditor)
```

> ⚠️ **`DEPENDS ${src}` 不跟踪 `#include`**。改了公共头，依赖它的 shader 不会重编。三种对策：
> 1. **depfile**：若编译器支持 `-MD -MF <file>`，用 `add_custom_command(... DEPFILE <file>)`（需要 Ninja 生成器）
> 2. **粗粒度**：`file(GLOB_RECURSE ... CONFIGURE_DEPENDS)` + 把 include 目录整体作为 `DEPENDS`
> 3. **自己扫**：构建前跑一个脚本，解析 `#include` 生成 depfile（推荐，可控）

### 1.3 方案 B：自研 Cooker（推荐给大项目）

CMake 只负责在构建前调一次 Python/C++ Cooker，Cooker 自己做内容哈希缓存与并行：

```cmake
find_package(Python3 COMPONENTS Interpreter REQUIRED)

add_custom_command(
    OUTPUT  ${CMAKE_CURRENT_BINARY_DIR}/shader_manifest.json
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/shader_cooker.py
            --src ${SHADER_SRC_DIR}
            --out ${SHADER_OUT_DIR}
            --jobs $<IF:$<BOOL(${CMAKE_BUILD_PARALLEL_LEVEL})>,${CMAKE_BUILD_PARALLEL_LEVEL},0>
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/shader_cooker.py
    COMMENT "Cooking shaders"
    VERBATIM
)

add_custom_target(ShaderCook ALL
    DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/shader_manifest.json)
add_dependencies(GameEditor ShaderCook)
```

| 维度 | CMake 方案 | 自研 Cooker |
|------|-----------|------------|
| 增量精度 | 依赖 depfile | **内容哈希，精确** |
| 变体管理 | 手写循环 | 声明式清单 |
| 去重 | 无 | ✅ 二进制去重 |
| 反射 + 代码生成 | 要额外 target | 一条流水线 |
| 并行 | CMake 自己做 | 自己控制（子进程/线程池） |
| 复杂度 | 低 | 中 |

> **推荐**：小项目用 CMake 方案；shader 数量 > 100 或变体 > 500 时上自研 Cooker。

### 1.4 可重现构建

哈希里**绝对不能**出现的：

- ❌ 绝对路径（`D:\proj\...`）
- ❌ 时间戳 / 构建序号
- ❌ 用户名 / 主机名
- ❌ 随机顺序的 map 遍历

```cpp
// ✅ 正确：规范化 + 排序
std::string CanonicalSourceHash(const ShaderKey& k) {
    std::string s;
    s += NormalizePath(k.source);          // 统一为相对路径 + '/' 分隔符
    s += '|';
    s += k.entryPoint;
    s += '|';
    auto kws = k.keywords;
    std::sort(kws.begin(), kws.end());     // ★ 排序
    for (auto& w : kws) { s += w; s += '|'; }
    s += std::to_string(k.stage);
    return blake3_hex(s);
}

std::string NormalizePath(std::string_view p) {
    std::string r(p);
    std::ranges::replace(r, '\\', '/');
    // 去掉盘符/前缀，保留相对仓库根目录的路径
    if (auto pos = r.find("/shaders/"); pos != std::string::npos)
        return r.substr(pos + 1);
    return r;
}
```

> **为什么重要**：可重现 = 缓存可跨机器共享 = CI 与本地能共用一份 shader 缓存，编译时间从分钟级降到秒级。

## 2. 工具链命令速查

### 2.1 编译

```bash
# ===== DXC：HLSL → SPIR-V =====
dxc -T ps_6_6 -E PSMain -spirv -fspv-target-env=vulkan1.3 \
    -Fo out.spv in.hlsl

# 常用附加选项
-fvk-b-shift 20 all          # CBV binding 偏移
-fvk-s-shift 10 all          # sampler
-fvk-t-shift 0  all          # SRV
-fvk-u-shift 30 all          # UAV
-fvk-use-dx-layout           # 按 HLSL 打包规则（与 D3D12 一致）
-fspv-reflect                # 生成反射辅助信息
-fspv-debug=vulkan-with-source   # 调试信息
-enable-16bit-types
-HV 2021                     # HLSL 2021（启用模板）
-Zi -Fd shader.pdb           # 生成 shader PDB
-I <include-dir>
-D NAME=1

# ===== glslang：GLSL → SPIR-V =====
glslangValidator -V --target-env vulkan1.3 -S frag -o out.spv in.frag
glslangValidator -V -H -o out.spv in.frag      # -H 输出人类可读

# ===== slangc：Slang → SPIR-V =====
slangc -target spirv -entry main -stage fragment -o out.spv in.slang

# ===== Clang + LLVM SPIR-V 后端：OpenCL → SPIR-V =====
clang --target=spirv64 -O1 -c kernel.cl -o kernel.spv
```

### 2.2 校验与优化

```bash
# 校验（必须！）
spirv-val --target-env vulkan1.3 out.spv

# 优化（Release）
spirv-opt -O --strip-nonsemantic -o opt.spv out.spv

# 优化（Debug：保留调试信息）
spirv-opt -O -g -o opt.spv out.spv

# 常用 opt pass
spirv-opt --legalize-hlsl out.spv -o out.spv   # HLSL 合法化（DXC 输出常用）
spirv-opt --strip-debug out.spv -o out.spv     # 剥 OpName/OpSource（会破坏名字反射！）
spirv-opt --strip-nonsemantic out.spv -o out.spv
spirv-opt --eliminate-dead-code-aggressive out.spv -o out.spv
spirv-opt --convert-relaxed-to-half out.spv -o out.spv

# 反汇编
spirv-dis out.spv -o out.spvasm

# 绑定重编号
spirv-remap --map all --dce all --opt all -o remapped.spv out.spv

# 崩溃最小化（驱动崩溃时用）
spirv-reduce out.spv --shrink-replace ...
```

### 2.3 反射

```bash
spirv-reflect out.spv            # 完整 dump
spirv-reflect out.spv > out.refl.txt
```

### 2.4 推荐的构建期流水线

```
源码 ──▶ dxc/glslang ──▶ spirv-val ──▶ spirv-opt ──▶ spirv-val ──┐
                            (校验1)                   (校验2)      │
                                                                   ▼
                                            spirv-dis (.txt) + spirv-reflect (.refl.txt)
                                                                   │
                                                                   ▼
                                                    自研反射 → .refl.bin + .h
```

> **为什么校验两次**：`spirv-opt` 有可能产出违反环境规则的模块（罕见但存在）。opt 之后必须再校验一次。

## 3. 产物组织

```
build/shaders/
├── gbuffer.frag.spv              # 编译产物
├── gbuffer.frag.spv.txt          # 反汇编（人肉排查）
├── gbuffer.frag.refl.txt         # spirv-reflect 输出（可 diff）
├── gbuffer.frag.refl.bin         # 引擎自研序列化（运行时加载）
└── manifest.json                 # key → {spv, refl, hash, deps}

generated/
└── shader_bindings.h             # C++ 绑定头
```

| 产物 | 是否提交 Git | 用途 |
|------|------------|------|
| `.spv` | 视情况（可加速 CI） | 运行时加载 |
| `.spv.txt` | ❌ | 排查用 |
| `.refl.txt` | ✅ **推荐提交** | **Code Review 时 diff 布局变更** |
| `.refl.bin` | ❌ | 运行时加载 |
| `manifest.json` | ✅ | 变体清单 |
| `shader_bindings.h` | ✅（若 C++ 依赖它编译） | 编译期契约 |

> **最有用的实践**：把 `.refl.txt` 提交进 Git。Reviewer 一眼就能看到"这次改动把 `PerFrame` 的 `cameraPos` 从 offset 64 挪到了 80"，而不是等运行时黑屏。

## 4. 调试

### 4.1 四类调试手段

| 手段 | 需要什么 | 能看到什么 |
|------|---------|-----------|
| **反汇编比对** | `spirv-dis` | SPIR-V 指令 |
| **反射 dump** | `spirv-reflect` | 绑定/布局 |
| **源码级断点** | NSDI 调试信息 + Nsight Graphics / RenderDoc | shader 源码单步 |
| **debugPrintf** | `VK_KHR_shader_non_semantic_info` + 校验层 | shader 内打印到调试输出 |

### 4.2 保留调试信息

```bash
# DXC
dxc -fspv-debug=vulkan-with-source -Zi -Fd shader.pdb -spirv ... -Fo out.spv in.hlsl

# LLVM SPIR-V 后端
llc -g --spirv-ext=+SPV_KHR_non_semantic_info in.ll -o out.spvt

# glslang
glslangValidator -V -gVS ... # 或 --amb 保留
```

> ⚠️ **矛盾点**：调试需要保留信息，发布需要剥离。解法是**两套产物**：
> - Debug 构建：`-O0` + 完整调试信息 + 不优化（便于单步）
> - Release 构建：`-O` + `--strip-nonsemantic`（保留 `OpName` 供反射！）
>
> **永远不要 strip `OpName`** 如果你的 set 分配依赖名字（见 [02 篇 §8.3](02-SPIRV-Reflect库详解与实战.md)）。

### 4.3 debugPrintf

**GLSL**（`GL_EXT_debug_printf`）：

```glsl
#extension GL_EXT_debug_printf : enable
void main() {
    debugPrintfEXT("uv=%f,%f albedo=%f", uv.x, uv.y, albedo.r);
}
```

**HLSL**（DXC 支持 `printf`，落到 NonSemantic）：

```hlsl
float4 PSMain(VSOut i) : SV_Target {
    printf("uv=(%f, %f)\n", i.uv.x, i.uv.y);
    return ...;
}
```

**启用**：

```cpp
// 1) 设备扩展
deviceExtensions.push_back(VK_KHR_SHADER_NON_SEMANTIC_INFO_EXTENSION_NAME);

// 2) 校验层配置（vkconfig 或代码）
VkValidationFeatureEnableEXT enables[] = { VK_VALIDATION_FEATURE_ENABLE_DEBUG_PRINTF_EXT };
VkValidationFeaturesEXT features{
    .sType = VK_STRUCTURE_TYPE_VALIDATION_FEATURES_EXT,
    .enabledValidationFeatureCount = 1,
    .pEnabledValidationFeatures = enables,
};
// 挂到 VkInstanceCreateInfo::pNext
```

> 输出会走调试回调（`VK_DEBUG_UTILS_MESSAGE_SEVERITY_INFO_BIT`）。Nsight Graphics 与 RenderDoc 也能显示。

### 4.4 图形调试器

| 工具 | 强项 |
|------|------|
| **RenderDoc** | 帧捕获、资源查看、shader 单步（需调试信息）、debugPrintf 显示 |
| **Nsight Graphics** | 源码级 shader 断点、性能分析、占用率/瓶颈定位 |
| **Radeon GPU Profiler** | AMD 平台深度分析（ISA 级） |
| **Intel GPA** | Intel 平台 |
| **Xcode Metal Debugger** | Metal 平台 |

**排查 "画面不对" 的标准流程**：

```
① RenderDoc 抓帧
② 看 Pipeline State → Descriptor Sets
   - 每个 binding 是否都绑定了？
   - 类型是否与反射一致？
③ 看 Shader 反射（工具内）→ 与你的 .refl.txt 比对
④ 在某像素上单步 → 观察变量
⑤ 用 debugPrintf 兜底（当单步不可用，如驱动不支持）
```

## 5. 性能优化

### 5.1 加载期

| 优化 | 做法 | 收益 |
|------|------|------|
| **持久化 PipelineCache** | `vkGetPipelineCacheData` → 存盘 → 下次 `VkPipelineCacheCreateInfo::pInitialData` | 消除首次启动的 PSO 编译 |
| **异步 PSO 创建** | 工作线程 + `VkPipelineCache` | 消除卡顿 |
| **离线预编译** | CI 上跑一遍，把 PipelineCache 数据随包发布 | 首启动即流畅 |
| **二进制去重** | 相同 SPIR-V 合并 | 减少磁盘与加载 |
| **反射结果序列化** | 构建期生成 `.refl.bin`，运行时不解析 SPIR-V | 加载提速 |

### 5.2 运行期

| 优化 | 做法 | 适用 |
|------|------|------|
| **Push Descriptor** | `VK_KHR_push_descriptor`：不分配描述符集，直接 `vkCmdPushDescriptorSet` | PerDraw 高频小数据 |
| **Descriptor Update Template** | `VK_KHR_descriptor_update_template`：把写入模板化，减少驱动开销 | 每帧大量相似更新 |
| **Bindless + UpdateAfterBind** | 全局数组 + 增量写 | 现代 GPU-Driven 主流 |
| **Descriptor Buffer** | `VK_EXT_descriptor_buffer`：描述符放进 GPU 缓冲，CPU 直写 | 新硬件，开销最低 |
| **动态偏移** | `UNIFORM_BUFFER_DYNAMIC` + `vkCmdBindDescriptorSets(pDynamicOffsets)` | PerDraw UBO，避免多套描述符集 |

### 5.3 Shader 侧

| 优化 | 说明 |
|------|------|
| 优先特化常量，少变体 | 见 [05 篇](05-可编程着色器元编程.md) |
| 用 `spirv-opt -O` | 内联 + 死代码消除 |
| 避免过度寄存器占用（uber shader 分支） | 影响 occupancy |
| 用 `NonUniformResourceIndex` | 正确且能避免驱动的保守处理 |
| 注意矩阵主序 | 反射告诉你实际值，别假设 |

## 6. CI 集成

### 6.1 检查项

```yaml
# .github/workflows/shaders.yml（示意）
jobs:
  shader-check:
    steps:
      - name: Cook shaders
        run: python tools/shader_cooker.py --src shaders --out build/shaders --jobs 0

      - name: Validate all SPIR-V
        run: |
          for f in build/shaders/*.spv; do
            spirv-val --target-env vulkan1.3 "$f" || exit 1
          done

      - name: Diff reflection dumps
        run: |
          git diff --exit-code -- shaders_ref/ || \
            (echo "::warning::Shader reflection changed — review carefully"; exit 0)

      - name: Report variant stats
        run: python tools/report_variants.py build/shaders/manifest.json

      - name: Smoke test (software Vulkan)
        run: ./build/RenderTests --headless --device lavapipe --frames 10
```

### 6.2 必做的四项 CI 检查

| 检查 | 为什么 |
|------|-------|
| `spirv-val` 全量 | 非法 SPIR-V 在驱动上行为未定义 |
| `.refl.txt` diff | 意外布局变更立刻可见 |
| 变体数量/体积报告 | 防止变体爆炸悄悄恶化 |
| 软件渲染冒烟 | 抓到"编译通过但画面崩"的问题 |

### 6.3 Shader 缓存共享

```
CI 产出 shader-cache/ (按内容哈希索引)
        │
        ├──▶ 上传到对象存储
        │
        └──▶ 开发者本地 --cache-from=remote
```

因为哈希是内容相关的、可重现的，本地与 CI 可以共享同一份缓存。

## 7. 故障排查清单

### 7.1 编译期

| 错误 | 原因 | 解法 |
|------|------|------|
| `error: unresolved external` | 函数只有声明 | 补定义或 `#include` |
| `error: cannot convert from X to Y` | 类型不匹配 | 检查模板/宏展开 |
| `OpEntryPoint interface list mismatch` | SPIR-V 版本规则 | 用 `-fspv-target-env=vulkan1.3`（产出 SPIR-V 1.6） |
| `spirv-val: DescriptorSet not decorated` | 资源缺 binding | 补 `[[vk::binding]]` 或 register |
| `spirv-val: entry point not found` | `-E` 名字错 | 检查入口点名 |
| 编译器崩溃 | 编译器 bug | `spirv-reduce` 最小化用例后报 issue |

### 7.2 反射期

| 错误码 | 含义 | 解法 |
|--------|------|------|
| `ERROR_SPIRV_INVALID_CODE_SIZE` | size 不是 4 的倍数 / 传了字数 | 传 `words * 4` |
| `ERROR_SPIRV_INVALID_MAGIC_NUMBER` | 不是 SPIR-V（可能是 DXIL 或文本） | 检查文件是否编成 SPIR-V |
| `ERROR_SPIRV_SET_NUMBER_OVERFLOW` | set ≥ 64 | 约束 shader 的 space 编号 |
| `ERROR_PARSE_FAILED` | 非法/损坏模块 | 先 `spirv-val` |
| `ERROR_ELEMENT_NOT_FOUND` | 精确查询未命中 | 检查 set/binding/名字 |
| `name` 为空 | 被 strip debug | 不要完整 strip |

### 7.3 运行期

| 现象 | 排查顺序 |
|------|---------|
| Pipeline 创建失败 | ① PushConstant 超 128B？② 描述符超设备上限？③ set 空洞未填 NULL？ |
| 校验层报类型不匹配 | 比对 `.refl.txt` 与你的写入代码 |
| 画面黑但无报错 | ① `COMBINED_IMAGE_SAMPLER` 是否同时写了 image+sampler？② 描述符集是否绑定？③ 图像布局是否正确？ |
| 数据错位 | ① `static_assert` 是否通过？② 矩阵主序？③ std140 padding？ |
| 偶发花屏 | ① 描述符写入后是否被另一帧覆盖（池轮转）？② `UpdateAfterBind` 缺少屏障？ |
| PSO 创建卡顿 | ① PipelineCache 是否持久化？② 是否预热？ |
| 热重载后崩溃 | ① 旧对象是否延迟销毁？② 布局变了是否重分配描述符？ |

### 7.4 一个诊断模板

遇到问题先跑这四步，90% 能定位：

```bash
# ① 合法性
spirv-val --target-env vulkan1.3 shader.spv

# ② 反射事实
spirv-reflect shader.spv > /tmp/refl.txt
cat /tmp/refl.txt

# ③ 与上一版 diff（若 .refl.txt 已提交）
git diff -- shaders_ref/shader.refl.txt

# ④ 反汇编看细节
spirv-dis shader.spv | grep -E "OpDecorate|OpMemberDecorate|OpEntryPoint"
```

## 8. 依赖与版本管理

| 依赖 | 建议 |
|------|------|
| **SPIRV-Reflect** | 源码直拷 + 固定 commit；CI 里加枚举对齐静态断言 |
| **SPIRV-Tools** | 与 Vulkan SDK 版本对齐（SDK 自带） |
| **SPIRV-Cross** | 若使用，固定 tag |
| **DXC** | 用官方 release 或 Vulkan SDK 自带；记录版本号 |
| **glslang** | Vulkan SDK 自带即可 |
| **Vulkan Headers** | 与目标 SDK 一致 |

**版本矩阵要记录**：

```json
// tools/shader_toolchain_versions.json
{
  "vulkan_sdk":   "1.3.290.0",
  "dxc":          "v1.8.2405",
  "glslang":      "14.0.0",
  "spirv_reflect":"<git sha>",
  "spirv_tools":  "2024.2",
  "spirv_cross":  "vulkan-sdk-1.3.290.0"
}
```

> **为什么**：不同版本的编译器会产出**不同布局**（尤其是矩阵主序与 std140 边界）。版本矩阵能解释"为什么换了台机器结果不一样"。

### 9.2 八条核心结论

| # | 结论 |
|---|------|
| 1 | **SPIR-V 是唯一真相**：换编译器不该改运行时代码 |
| 2 | **反射只给事实，语义靠约定**：set/binding/offset 是事实，"PerFrame" 是约定 |
| 3 | **名字不可靠**：依赖 `OpName`，会被 strip；关键逻辑不要依赖它 |
| 4 | **反射前先校验**：`spirv-val` 是必需的，不是可选的 |
| 5 | **改写后必须用新 SPIR-V**：`Change*` 后一律 `spvReflectGetCode()` |
| 6 | **能特化就别变体**：特化常量是治理变体爆炸的性价比之王 |
| 7 | **代码生成把契约钉进编译期**：`static_assert` 让 shader 改动在 C++ 编译期暴露 |
| 8 | **抽象层隔离反射库**：换库只改一个文件 |

### 9.3 落地路线图（给刚起步的团队）

| 阶段 | 做什么 | 产出 |
|------|-------|------|
| **P0** | 引入 SPIRV-Reflect，跑通"加载 → 枚举 → 打印" | 能看见 shader 的绑定表 |
| **P1** | 用反射生成 `DescriptorSetLayout` / `PipelineLayout` | 不再手写描述符布局 |
| **P2** | 加入 `spirv-val` + `spirv-opt` 到构建 | 编译期拦截非法 shader |
| **P3** | 生成 C++ 绑定头 + `static_assert` | 双重维护问题消失 |
| **P4** | 命名约定 → set 自动分配 | 语义自动化 |
| **P5** | 布局哈希缓存 + Bindless | 描述符开销下降 |
| **P6** | 变体 Cooker + 二进制去重 + PSO 预热 | 构建与加载提速 |
| **P7** | 热重载 | 迭代速度质变 |
